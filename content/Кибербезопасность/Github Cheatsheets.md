### Инициализация и копирование
1. git init (создать новый локальный репозиторий)
2. git clone (URL) (склонировать репозиторий)
3. git clone --depth 1 (URL) (клонировать только последний коммит)

### Добавление и коммиты
1. git status                    # посмотреть изменения
2. git add (файл)                # добавить файл в индекс
3. git add .                     # добавить все изменения (кроме игнорируемых)
4. git add -A                    # то же, что и `git add .`, но включая удалённые файлы
5. git reset (файл)              # убрать файл из индекса (но оставить изменения)
6. git commit -m "Сообщение"     # сделать коммит
7. git commit -am "Сообщение"    # добавить и закоммитить отслеживаемые файлы (без add)

### Синхронизация с удалённым репозиторием
1. git remote -v                 # посмотреть привязанные удалённые репозитории
2. git remote add origin (URL)   # добавить удалённый репозиторий
3. git push -u origin main       # отправить ветку main на GitHub (первый раз)
4. git push                      # отправить изменения (после первого пуша)
5. git pull                      # забрать и слить изменения с удалённого репозитория

### Ветки (branches)
1. git branch                    # список локальных веток (* = текущая)
2. git branch -a                 # все ветки (локальные + удалённые)
3. git checkout <ветка>          # переключиться на ветку
4. git switch <ветка>            # современная альтернатива checkout (только для веток)
5. git checkout -b <новая>       # создать и переключиться на новую ветку
6. git switch -c <новая>         # то же самое, но через switch
7. git branch -d <ветка>         # удалить ветку (только если слита)
8. git push origin --delete <ветка>  # удалить ветку на GitHub

### Просмотр истории и изменений
1. git log                       # история коммитов (q — выйти)
2. git log --oneline             # краткий журнал
3. git log -p                    # с diff'ами
4. git diff                      # изменения в рабочей директории (ещё не в индексе)
5. git diff --staged             # изменения, добавленные в индекс (готовые к коммиту)
6. git show <хеш>                # посмотреть конкретный коммит

### Отмена изменений
1. git restore <файл>            # отменить изменения в файле (в рабочей директории)
2. git restore --staged <файл>   # убрать файл из индекса (аналог git reset)
3. git reset --hard HEAD         # отменить ВСЕ локальные изменения (осторожно!)
4. git revert <хеш>              # создать новый коммит, отменяющий старый (без перезаписи истории)

### Работа с SSH
1. ssh-keygen -t ed25519 -C "email@example.com"   # сгенерировать ключ
2. cat ~/.ssh/id_ed25519.pub                      # показать публичный ключ
3. eval "$(ssh-agent -s)"                         # запустить агент
4. ssh-add ~/.ssh/id_ed25519                      # добавить ключ
